{{- $src := . -}} {{- $result := "" -}} {{- if or (strings.HasPrefix $src "http://") (strings.HasPrefix $src "https://") (strings.HasPrefix $src "data:") -}} {{- $result = $src -}} {{- else -}} {{- $clean := strings.TrimPrefix "/" $src -}} {{- $basePath := strings.TrimPrefix "/" (urls.Parse site.BaseURL).Path -}} {{- $basePath = strings.TrimSuffix "/" $basePath -}} {{- if and (ne $basePath "") (strings.HasPrefix $clean (printf "%s/" $basePath)) -}} {{- $result = printf "/%s" $clean -}} {{- else -}} {{- $result = $clean | relURL -}} {{- end -}} {{- end -}} {{- return $result -}}
